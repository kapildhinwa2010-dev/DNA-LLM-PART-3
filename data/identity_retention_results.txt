import torch.nn.functional as F
from transformers import AutoModelForCausalLM, AutoTokenizer

try: torch.set_num_threads(max(1, (os.cpu_count() or 1) - 1))
except: pass

t0 = time.time()
print("=" * 80)
print("🧬 MISSION 1.23: CONTRASTIVE IDENTITY DNA")
print("=" * 80)

model = AutoModelForCausalLM.from_pretrained("Qwen/Qwen2.5-0.5B-Instruct", dtype=torch.float32, low_cpu_mem_usage=True)
tok = AutoTokenizer.from_pretrained("Qwen/Qwen2.5-0.5B-Instruct")
tok.padding_side = "left"
if tok.pad_token is None: tok.pad_token = tok.eos_token
model.eval()
NL = model.config.num_hidden_layers
D = model.config.hidden_size

# Calibration with Explicit Entities
cal_prompts = [
    "The capital of France is", "The capital of Germany is",
    "The capital of Italy is", "The capital of Spain is",
    "The capital of China is", "The capital of India is",
    "The president of USA in 2008 was", "The chemical symbol for gold is",
]

eval_prompts = ["The capital of Japan is", "The capital of Mexico is", "The chemical symbol for silver is"]
eval_labels = ["Japan", "Mexico", "Silver"]

RANK = 16
IDENTITY_SLOTS = 4  # Reserve top 4 dimensions for Entity Identity

boxes = {l: {} for l in range(NL)}
def make_cap(l):
    def hook(m, i, o):
        boxes[l]["out"] = o.detach().float()
        return o
    return hook

cal_inp = tok(cal_prompts, return_tensors="pt", padding=True, truncation=True, max_length=24)
cal_mask = cal_inp["attention_mask"].bool()

hooks = [model.model.layers[l].register_forward_hook(make_cap(l)) for l in range(NL)]
with torch.no_grad(): model(**cal_inp)
for h in hooks: h.remove()

print("📐 Building Contrastive DNA Bases (Identity + SVD)...")
bases = {}
for l in range(NL):
    out_states = boxes[l]["out"][cal_mask]  # (N, D)
    mu = out_states.mean(0)
    centered = out_states - mu
    
    # 1. SVD for Grammar/Logic (Normal DNA)
    _, _, Vt = torch.linalg.svd(centered, full_matrices=False)
    svd_dirs = Vt[:(RANK - IDENTITY_SLOTS)]
    
    # 2. Contrastive Identity Directions (Forced DNA)
    # Take differences between random pairs of calibration states to simulate entity contrast
    # (In a real system, we'd tag entity tokens. Here we use random prompt differences as proxy)
    # Better: Take the last token of each prompt (which holds the aggregated context)
    last_idx = cal_mask.sum(dim=1) - 1
    prompt_states = torch.stack([boxes[l]["out"][b, last_idx[b]] for b in range(len(cal_prompts))])
    
    # Create contrastive directions: Prompt[i] - Prompt[j]
    contrast_dirs = []
    for i in range(min(4, len(cal_prompts))):
        for j in range(i+1, min(4, len(cal_prompts))):
            diff = prompt_states[i] - prompt_states[j]
            diff = diff - mu
            if diff.norm() > 1e-6:
                contrast_dirs.append(diff / diff.norm())
    
    if len(contrast_dirs) >= IDENTITY_SLOTS:
        contrast_dirs = torch.stack(contrast_dirs[:IDENTITY_SLOTS])
    else:
        # Fallback if not enough pairs
        contrast_dirs = torch.randn(IDENTITY_SLOTS, D)
        contrast_dirs = F.normalize(contrast_dirs, dim=1)
    
    # 3. Gram-Schmidt Orthogonalization (Combine Identity + SVD without overlap)
    # Start with identity directions
    basis = [contrast_dirs[0]]
    for i in range(1, IDENTITY_SLOTS):
        v = contrast_dirs[i]
        for b in basis:
            v = v - (v @ b) * b
        if v.norm() > 1e-6:
            basis.append(v / v.norm())
            
    # Add SVD directions
    for i in range(svd_dirs.shape[0]):
        v = svd_dirs[i]
        for b in basis:
            v = v - (v @ b) * b
        if v.norm() > 1e-6:
            basis.append(v / v.norm())
        if len(basis) >= RANK:
            break
            
    Vk = torch.stack(basis[:RANK])
    bases[l] = (Vk.to(torch.float32), mu)

print(f"✅ Contrastive Bases ready ({time.time()-t0:.1f}s)")

# Hook
STATE = {"dna": False}
def make_hook(l_idx):
    def hook(m, i, o):
        if not STATE["dna"]: return o
        Vk, mu = bases[l_idx]
        out = o.float()
        flat = out.reshape(-1, D)
        centered = flat - mu
        projected = (centered @ Vk.T) @ Vk + mu
        return projected.reshape(o.shape).to(o.dtype)
    return hook

dna_hooks = [model.model.layers[l].register_forward_hook(make_hook(l)) for l in range(NL)]

# Eval
def get_flow(ids, dna=False):
    STATE["dna"] = dna
    with torch.no_grad():
        out = model(ids, output_hidden_states=True)
    H = torch.stack([out.hidden_states[l+1][0, -1, :].float() for l in range(NL)], dim=0)
    logits = out.logits[0, -1].float()
    probs = F.softmax(logits, dim=-1)
    tk = torch.topk(probs, 5)
    STATE["dna"] = False
    return H, tk.indices, tk.values

def teacher_seqs(prompt, steps):
    ids = tok(prompt, return_tensors="pt").input_ids
    seqs = [ids.clone()]
    with torch.no_grad():
        for _ in range(steps):
            out = model(ids)
            nxt = out.logits[:, -1, :].argmax(-1, keepdim=True)
            ids = torch.cat([ids, nxt], dim=-1)
            seqs.append(ids.clone())
    return seqs

seqs_list = [teacher_seqs(p, 2) for p in eval_prompts]
orig_flows = [[get_flow(ids, dna=False) for ids in seqs] for seqs in seqs_list]

print("\n" + "=" * 80)
print("🧪 CONTRASTIVE DNA — IDENTITY COLLAPSE TEST")
print("=" * 80)

for st in range(3):
    step_recs = []
    for pi, seqs in enumerate(seqs_list):
        H_o, top_o, _ = orig_flows[pi][st]
        H_d, top_d, _ = get_flow(seqs[st], dna=True)
        step_recs.append((eval_labels[pi], H_o, H_d))
        
    print(f"\n--- Step {st} ---")
    for i in range(len(step_recs)):
        for j in range(i+1, len(step_recs)):
            n1, H1_o, H1_d = step_recs[i]
            n2, H2_o, H2_d = step_recs[j]
            
            od = 1.0 - F.cosine_similarity(H1_o.reshape(1,-1), H2_o.reshape(1,-1)).item()
            dd = 1.0 - F.cosine_similarity(H1_d.reshape(1,-1), H2_d.reshape(1,-1)).item()
            ratio = dd / od if od > 1e-4 else 1.0
            
            v = "🟢 OK" if ratio >= 0.70 else ("🟡 PARTIAL" if ratio >= 0.40 else "🔴 COLLAPSED")
            print(f"{n1}-{n2:<10} Orig={od:.4f} DNA={dd:.4f} Ratio={ratio:.3f} {v}")

print("\n" + "=" * 80)
print("🏆 MISSION 1.23 VERDICT")
print("=" * 80)
print("Agar Japan-Mexico ratio 🟢 OK (>=0.70) aaya:")
print("🎉 EUREKA! Contrastive DNA ne identity collapse ko tod diya!")
print("   Humne SVD ko force kiya ki wo 'Entity Differences' ko Rank-16 mein include kare.")
print("   Yeh REAL Reverse DNA hai jo production mein use ho sakta hai.")

64================================================================
REVERSE DNA PROJECT — COMPLETE PROBLEM STATEMENT
================================================================

GOAL:
Compress a 0.5B parameter LLM (Qwen2.5-0.5B-Instruct) into a tiny 
"mathematical DNA" file (~KB range) that preserves the model's 
knowledge and behavior. The DNA should be a set of basis vectors 
(functions) per layer that can reconstruct the model's internal 
representations during inference.

MODEL: Qwen/Qwen2.5-0.5B-Instruct
ARCHITECTURE: 24 layers, 896 hidden dimensions, 151936 vocab size
HARDWARE: CPU only (no GPU)

================================================================
WHAT WE TRIED (CHRONOLOGICAL)
================================================================

PHASE 1: DELTA DNA (Compress layer shifts)
-------------------------------------------
Idea: For each layer, compute delta = layer_output - layer_input.
Build Rank-16 SVD basis from calibration prompts. During generation,
project each layer's delta onto this basis and reconstruct.

Result: 
- Internal shift direction preserved ~96% (Logit Lens test)
- BUT final text generation accuracy dropped to ~23%
- Identity collapse: Japan and Mexico representations merged (ratio 0.001)
- Autoregressive generation diverges after Step 0 (Butterfly Effect)

PHASE 2: MULTI-LAYER DNA SWEEP
-------------------------------
Idea: Compress 1 layer vs 7 layers vs all 24 layers simultaneously.

Result:
- 1 layer: ~23% accuracy
- 7 layers: ~23% accuracy  
- 24 layers: ~17-20% accuracy (WORSE)
- Multi-layer compression made things worse, not better.

PHASE 3: LOGIT LENS / INTERNAL BEHAVIOR MEASUREMENT
----------------------------------------------------
Idea: Measure if internal "thinking direction" is preserved even when
final tokens are wrong. Multiply layer delta by LM Head matrix to get
"logit contribution" per layer.

Result:
- Average shift preservation across all 24 layers: 96.4%
- Late layers: 99%+ preserved
- CONCLUSION: Internal knowledge IS preserved. The problem is in 
  how compressed states interact during autoregressive generation.

PHASE 4: RESCUE ATTEMPTS (Post-hoc fixes)
------------------------------------------
Tried: Linear rescue matrix, MLP rescue, Concept vector injection,
Top-K oracle rescue, Nearest-neighbor concept bank, Trajectory-aware
rescue, Multi-scale spiral transforms.

Result: ALL FAILED for standalone generation.
- Oracle-assisted rescue (using original model's logits) worked perfectly
- But standalone DNA could not self-correct
- Root cause: 24 layers of non-linearity entangle compression errors
  in ways that cannot be undone by post-hoc linear/non-linear maps

PHASE 5: ANTS-BIL / CEMENT FLOW MAPPING
-----------------------------------------
Idea: Like filling an ant hill with cement to map all tunnels, inject
traceable signals through the model to map knowledge routing paths.

Result:
- Discovered IDENTITY COLLAPSE: Japan-Mexico original distance = 0.1119,
  but after DNA compression distance = 0.0001 (ratio 0.001)
- Layer 11 cement signature showed Japan and Mexico active dimensions
  became identical after compression
- Early layers (0-7) drift starts immediately when new tokens are generated

PHASE 6: SELECTIVE DEPTH DNA
-----------------------------
Idea: Keep early layers (0-7) uncompressed, only compress layers 8-23.

Result:
- Slight improvement in min_ratio (0.000 → 0.013)
- But identity collapse still severe
- CONCLUSION: Problem is not just in early layers, it's in the 
  compression strategy itself

PHASE 7: SPIRAL DNA (Non-linearity handling)
---------------------------------------------
Tried 4 interpretations:
1. Random Fourier Features (coordinate transform)
2. Phase encoding (direction only)
3. Log-unwinding (inverse twist)
4. Multi-scale chirp probes
5. All 4 combined

Result: ALL gave same ~0.001 identity ratio. No improvement.
CONCLUSION: Problem is not non-linearity per se, it's the compression
strategy destroying low-variance identity information.

PHASE 8: SPIRAL ZOOM (Multi-scale residual SVD)
-------------------------------------------------
Idea: Iteratively compress residuals. Level 1: Rank-16 on delta.
Level 2: Rank-8 on residual. Level 3: Rank-4 on residual. Etc.
This should capture low-frequency identity components.

Result: Adding zoom levels (16→8→4→2→1) gave NO improvement.
All configs: ~0.001 identity ratio.
CONCLUSION: The basis vectors simply don't contain entity-specific
directions because calibration set doesn't have enough entity diversity,
OR the identity information is not in the delta subspace at all.

PHASE 9: CALIBRATION COVERAGE TEST
------------------------------------
Idea: Include Japan/Mexico/Silver IN the calibration set. Does that fix
identity collapse?

Result:
- Japan-Mexico ratio improved from 0.001 to 0.058 (still COLLAPSED)
- Japan-Silver ratio improved to 0.839 (OK!)
- Mexico-Silver ratio improved to 0.745 (OK!)
- But truly unseen concepts (Brazil/Germany) still collapsed
- CONCLUSION: Calibration coverage helps partially but doesn't solve
  the fundamental problem. Japan-Mexico are too similar in the 
  template space.

PHASE 10: SANITY SWEEP — DIRECT IDENTITY RETENTION
----------------------------------------------------
Idea: Pure offline math test. No generation. Just project identity
vectors (Japan - France) through SVD basis and measure retention.

Result:
- Rank-16 average identity retention: 56.26%
- Full-rank (198) average identity retention: 73.26%
- Layer 11 at Rank-16: only 36.1% identity retention
- VERDICT: Even full-rank basis doesn't fully contain identity directions.
  Identity may be in embeddings/token path, not in delta subspace.

PHASE 11: DELTA vs ABSOLUTE STATE
-----------------------------------
Idea: Maybe knowledge is in absolute hidden states, not in deltas.
Test identity retention using SVD on absolute states vs deltas.

Result:
- DELTA retention at Rank-16: 18-24% (all layers)
- ABS_OUT retention at L5: 99.4%!
- ABS_OUT retention at L11: 99.3%!
- ABS_OUT retention at L17: 98.3%!
- BUT: This was a MATHEMATICAL ILLUSION. Only 8 calibration prompts
  were used, so Rank-16 SVD on 8 vectors = full rank = no compression.

PHASE 12: RESIDUAL STREAM DNA (Absolute state compression in generation)
-------------------------------------------------------------------------
Idea: Project absolute output states (not deltas) during generation.

Result:
- Japan-Mexico ratio: 0.000 (STILL COLLAPSED)
- Japan-Silver ratio: 0.637 (improved from 0.001)
- Mexico-Silver ratio: 0.571 (improved from 0.001)
- CONCLUSION: Absolute state helps for LARGE differences (Japan vs Silver)
  but SMALL differences (Japan vs Mexico, distance=0.1119) still collapse.

PHASE 13: CONTRASTIVE IDENTITY DNA
------------------------------------
Idea: Force entity-contrastive directions into the DNA basis.
Reserve top 4 slots for entity differences, fill rest with SVD.

Result:
- Japan-Mexico ratio: 0.000 (STILL COLLAPSED at all steps)
- Japan-Silver ratio: 0.478 (partial)
- CONCLUSION: Contrastive directions from last-token states are too weak.
  The entity identity signal gets washed out when ALL tokens are projected.

================================================================
CORE PROBLEMS IDENTIFIED
================================================================

PROBLEM 1: TEMPLATE vs ENTITY VARIANCE IMBALANCE
-------------------------------------------------
"The capital of [ENTITY] is" template has HUGE variance.
Entity identity (Japan vs Mexico vs France) has TINY variance.
Rank-16 SVD captures the template, crushes the entity.
Japan-Mexico original distance is only 0.1119, while Japan-Silver
is 0.3111. Small differences get rounded to zero in low-rank subspace.

PROBLEM 2: SINGLE BASIS FOR ALL TOKENS
----------------------------------------
We use ONE Rank-16 basis for ALL tokens (The, capital, of, Japan, is).
But "The", "capital", "of", "is" are identical across Japan/Mexico prompts.
Only "Japan" vs "Mexico" differs. When all tokens share the same basis,
the common tokens dominate and the entity token's unique signal is lost.

PROBLEM 3: IDENTITY NOT IN DELTA SUBSPACE
-------------------------------------------
Mission 1.20 showed that even full-rank delta basis only retains 73%
identity. This suggests entity identity is stored in the EMBEDDING layer
or early token representations, NOT in the layer deltas (residual shifts).
Compressing deltas fundamentally cannot preserve what isn't there.

PROBLEM 4: AUTOREGRESSIVE BUTTERFLY EFFECT
--------------------------------------------
Even if Step 0 is perfect, generating Step 1 introduces a tiny error.
This error becomes the INPUT for Step 1, which amplifies it.
By Step 2-3, the model is in a completely different region of the
manifold (Out-of-Distribution). No post-hoc correction can fix this.

PROBLEM 5: NON-LINEAR ENTANGLEMENT
------------------------------------
24 layers of SiLU, attention softmax, and LayerNorm entangle the
compression error with the actual knowledge in non-linear ways.
This "baked cake" cannot be unbaked by any linear or simple non-linear
post-hoc transformation.

PROBLEM 6: CALIBRATION GENERALIZATION GAP
-------------------------------------------
DNA bases built from calibration prompts cannot generalize to unseen
concepts. Even with diverse calibration (37 prompts), truly unseen
entities (Brazil, Germany) still collapse. The basis only knows
directions it has seen.

================================================================
UNEXPLORED DIRECTIONS
================================================================

1. TOKEN-CONDITIONED DNA:
   Use different bases for entity tokens vs grammar tokens.
   Don't compress entity tokens, or give them higher rank.

2. RESIDUAL-ONLY COMPRESSION:
   Extract the common template ("The capital of ___ is").
   Compress only the template. Leave entity-specific residual untouched.

3. EMBEDDING-LEVEL DNA:
   Since identity may be in embeddings, compress the embedding matrix
   instead of layer activations.

4. SPARSE / SELECTIVE COMPRESSION:
   Identify which tokens/dimensions carry identity (via attention weights
   or gradient attribution). Only compress non-identity components.

5. LEARNED COMPRESSION (Distillation):
   Train a small adapter network to map compressed states back to 
   original states. Requires GPU but may be the only real solution.

6. CONTRASTIVE LEARNING:
   Train the compression basis with a contrastive loss that explicitly
   penalizes entity collapse (Japan-Mexico distance must be preserved).

7. HIGHER RANK WITH PRUNING:
   Use Rank-64 or Rank-128, then prune less important directions.
   May preserve identity at cost of larger DNA file.

8. LAYER-WISE ADAPTIVE RANK:
   Give more rank to layers that carry more identity information
   (e.g., Layer 11 gets Rank-64, Layer 0 gets Rank-8).

THE FUNDAMENTAL QUESTION
================================================================

Can a 0.5B parameter LLM's knowledge be compressed into a ~KB-sized
mathematical representation (DNA) that preserves entity-level identity
AND supports autoregressive generation without drift?

Or is entity identity fundamentally non-compressible below a certain
rank threshold, making "Reverse DNA" impossible for fine-grained
knowledge preservation?
